--drop table DDL_Audit_Log
CREATE TABLE dbo.DDL_Audit_Log
(
    AuditID        BIGINT IDENTITY(1,1) PRIMARY KEY,
    EventTime      DATETIME2(3),
    EventType      NVARCHAR(200),
    DatabaseName   SYSNAME,
    SchemaName     SYSNAME NULL,
    ObjectName     SYSNAME NULL,
    ObjectType     NVARCHAR(100),
    TSQLCommand    NVARCHAR(MAX),
    LoginName      SYSNAME,
    UserName       SYSNAME,
    HostName       SYSNAME,
    Application    SYSNAME,
    ClientIP       NVARCHAR(48)
);
GO

USE master;
GO

CREATE OR ALTER TRIGGER trg_Server_DDL_Audit
ON ALL SERVER
FOR
    CREATE_DATABASE, ALTER_DATABASE, DROP_DATABASE,
    CREATE_LOGIN, ALTER_LOGIN, DROP_LOGIN,
    CREATE_SERVER_ROLE, ALTER_SERVER_ROLE, DROP_SERVER_ROLE
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @EventData XML = EVENTDATA();

    IF OBJECT_ID('master.dbo.DDL_Audit_Log','U') IS NOT NULL
    BEGIN
        INSERT INTO master.dbo.DDL_Audit_Log
        (
            EventTime,
            EventType,
            DatabaseName,
            ObjectName,
            ObjectType,
            TSQLCommand,
            LoginName,
            UserName,
            HostName,
            Application,
            ClientIP
        )
        SELECT
            SYSDATETIME(),
            @EventData.value('(/EVENT_INSTANCE/EventType)[1]', 'NVARCHAR(200)'),
            @EventData.value('(/EVENT_INSTANCE/DatabaseName)[1]', 'SYSNAME'),
            @EventData.value('(/EVENT_INSTANCE/ObjectName)[1]', 'SYSNAME'),
            @EventData.value('(/EVENT_INSTANCE/ObjectType)[1]', 'NVARCHAR(100)'),
            @EventData.value('(/EVENT_INSTANCE/TSQLCommand/CommandText)[1]', 'NVARCHAR(MAX)'),
            ORIGINAL_LOGIN(),
            NULL,
            HOST_NAME(),
            APP_NAME(),
            CONVERT(NVARCHAR(48), CONNECTIONPROPERTY('client_net_address'));
    END
END;
GO
