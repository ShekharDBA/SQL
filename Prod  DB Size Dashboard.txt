WITH DuplicateRows AS
(
    SELECT ROW_NUMBER() OVER
    (
        PARTITION BY Database_Name, SchemaName, Table_Name, CAST(Collection_time AS DATE)
        ORDER BY UsedSpaceMB DESC
    ) AS RowNum
	, Collection_time,database_name,table_name,SchemaName
    , rows,TotalSpaceMB, UsedSpaceMB,FreeSpaceMB,DataSizeMB,IndexSizeMB,TotalSizeGB,UsedSizeGB
    FROM [IVPProdAudit].[dbo].[AllDbTableSize]
    WHERE CAST(Collection_time AS DATE) = CAST(GETDATE() AS DATE)
)
SELECT COUNT(*) AS DuplicateCount
FROM DuplicateRows
WHERE RowNum > 1;

WITH DuplicateRows AS
(
    SELECT 
        ROW_NUMBER() OVER
        (
            PARTITION BY Database_Name, SchemaName, Table_Name, CAST(Collection_time AS DATE)
            ORDER BY UsedSpaceMB DESC
        ) AS RowNum,
        *
    FROM [IVPProdAudit].[dbo].[AllDbTableSize]
    WHERE CAST(Collection_time AS DATE) = CAST(GETDATE() AS DATE)
)
DELETE FROM DuplicateRows
WHERE RowNum > 1;
