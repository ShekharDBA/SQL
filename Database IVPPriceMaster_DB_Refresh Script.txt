
*********************** DB Size **********************************************

SELECT 
    name AS DatabaseName,
    CAST(SUM(size) * 8 / 1024 AS DECIMAL(10,2)) AS SizeMB,
	CAST(SUM(size) * 8.0 / 1024 / 1024 AS DECIMAL(10,2)) AS SizeGB
FROM sys.master_files
where  name in ('IVPPriceMaster','IVPPriceMaster_archive','IVPPriceMasterVendor','IVPPriceMasterVendor_Archive')
GROUP BY name
ORDER BY SizeMB DESC;
