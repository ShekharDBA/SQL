
SELECT 
    dp.name AS UserName,
    dp.type_desc AS UserType,
    dp.authentication_type_desc AS AuthType,
    dp.create_date,
    dp.modify_date
FROM sys.database_role_members drm
JOIN sys.database_principals dp 
    ON drm.member_principal_id = dp.principal_id
JOIN sys.database_principals rp 
    ON drm.role_principal_id = rp.principal_id
WHERE rp.name = 'db_datawriter'
ORDER BY dp.name;


SELECT 
    DB_NAME() AS DatabaseName,
    dp.name AS UserName,
    dp.type_desc AS UserType,
	rp.name
FROM sys.database_role_members drm
INNER JOIN sys.database_principals dp 
    ON drm.member_principal_id = dp.principal_id
INNER JOIN sys.database_principals rp 
    ON drm.role_principal_id = rp.principal_id
WHERE rp.name = 'db_datawriter'
ORDER BY dp.name;

DECLARE @DB SYSNAME, @SQL NVARCHAR(MAX);

DECLARE DB_CURSOR CURSOR FOR
SELECT DBName FROM @DBList;

OPEN DB_CURSOR;
FETCH NEXT FROM DB_CURSOR INTO @DB;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @SQL = '
    IF DB_ID(''' + @DB + ''') IS NOT NULL
    BEGIN
        INSERT INTO #DataWriterAccessDetails
        SELECT 
            ''' + @DB + ''' AS DatabaseName,
            dp.name AS UserName,
            dp.type_desc AS UserType,
            rp.name AS DataWriter
        FROM ' + QUOTENAME(@DB) + '.sys.database_role_members drm
        INNER JOIN ' + QUOTENAME(@DB) + '.sys.database_principals dp 
            ON drm.member_principal_id = dp.principal_id
        INNER JOIN ' + QUOTENAME(@DB) + '.sys.database_principals rp 
            ON drm.role_principal_id = rp.principal_id
        WHERE rp.name = ''db_datawriter''
        ORDER BY dp.name;
    END;
    ';
	--print @SQL;    
    EXEC sys.sp_executesql @SQL;

    FETCH NEXT FROM DB_CURSOR INTO @DB;
END;

CLOSE DB_CURSOR;
DEALLOCATE DB_CURSOR;

SELECT 'NPGPRD-IVPSDB03' AS ServerName,* FROM #DataWriterAccessDetails ORDER BY DatabaseName, UserName;
